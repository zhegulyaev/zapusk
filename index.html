Инструкция по настройке

1. Открой Google Таблицы → Расширения → Apps Script

2. Удали весь код и вставь этот:

function doPost(e) { try { var data = JSON.parse(e.postData.contents); var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName('FocusFlow'); if (!sheet) sheet = ss.insertSheet('FocusFlow'); sheet.clearContents(); sheet.getRange(1, 1).setValue(JSON.stringify(data)); return ContentService .createTextOutput(JSON.stringify({status:'ok'})) .setMimeType(ContentService.MimeType.JSON); } catch(err) { return ContentService .createTextOutput(JSON.stringify({status:'error',message:err.toString()})) .setMimeType(ContentService.MimeType.JSON); } } function doGet(e) { try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName('FocusFlow'); if (!sheet) return ContentService .createTextOutput(JSON.stringify({status:'empty'})) .setMimeType(ContentService.MimeType.JSON); var val = sheet.getRange(1,1).getValue(); if (!val || val==='') return ContentService .createTextOutput(JSON.stringify({status:'empty'})) .setMimeType(ContentService.MimeType.JSON); return ContentService .createTextOutput(val) .setMimeType(ContentService.MimeType.JSON); } catch(err) { return ContentService .createTextOutput(JSON.stringify({status:'error',message:err.toString()})) .setMimeType(ContentService.MimeType.JSON); } }

3. Нажми Развернуть → Новое развёртывание

4. Тип: Веб-приложение

5. Выполнять от имени: Я

6. Доступ: Все (не "вошедшие в Google")

7. Нажми Развернуть, разреши доступ

8. Скопируй URL и вставь в поле выше ☝️
